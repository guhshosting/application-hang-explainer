cmake_minimum_required(VERSION 3.24)
project(application-hang-explainer VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows with MSVC is required")
endif()
option(AHE_ANALYZE "Enable MSVC analysis" OFF)
option(AHE_ASAN "Enable AddressSanitizer" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(ahe_core src/model.cpp src/cli.cpp src/process.cpp src/threads.cpp src/wct.cpp src/inspection.cpp src/windows_error.cpp src/format.cpp)
target_include_directories(ahe_core PUBLIC include)
target_compile_definitions(ahe_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(ahe_core PUBLIC /W4 /WX /utf-8 /permissive- /EHsc)
target_link_libraries(ahe_core PUBLIC advapi32)
if(AHE_ANALYZE)
 target_compile_options(ahe_core PUBLIC /analyze)
endif()
if(AHE_ASAN)
 target_compile_options(ahe_core PUBLIC /fsanitize=address /Zi)
 target_link_options(ahe_core PUBLIC /INCREMENTAL:NO)
 get_filename_component(AHE_COMPILER_DIRECTORY "${CMAKE_CXX_COMPILER}" DIRECTORY)
 if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(AHE_ASAN_ARCH x86_64)
 else()
  set(AHE_ASAN_ARCH i386)
 endif()
 set(AHE_ASAN_RUNTIME "${AHE_COMPILER_DIRECTORY}/clang_rt.asan_dynamic-${AHE_ASAN_ARCH}.dll")
 if(NOT EXISTS "${AHE_ASAN_RUNTIME}")
  message(FATAL_ERROR "MSVC ASan runtime not found beside compiler; install the ASan component")
 endif()
endif()
add_executable(application-hang-explainer src/main.cpp)
target_link_libraries(application-hang-explainer PRIVATE ahe_core)
target_link_options(application-hang-explainer PRIVATE /MAP)
include(CTest)
if(BUILD_TESTING)
 add_executable(ahe_tests tests/tests.cpp)
 target_link_libraries(ahe_tests PRIVATE ahe_core)
 add_test(NAME synthetic COMMAND ahe_tests)
 add_test(NAME live COMMAND ahe_tests live)
 set_tests_properties(synthetic live PROPERTIES TIMEOUT 40)
 add_test(NAME help COMMAND application-hang-explainer --help)
 add_test(NAME version COMMAND application-hang-explainer --version)
 add_executable(ahe_integration tests/integration.cpp)
 target_link_libraries(ahe_integration PRIVATE ahe_core)
 add_test(NAME integration COMMAND ahe_integration)
 set_tests_properties(integration PROPERTIES TIMEOUT 40)
 add_test(NAME json_independent COMMAND ${CMAKE_COMMAND} -DTEST_EXE=$<TARGET_FILE:ahe_tests> -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/json.cmake)
 set_tests_properties(json_independent PROPERTIES TIMEOUT 15)
endif()
if(AHE_ASAN)
 foreach(AHE_TARGET application-hang-explainer ahe_tests ahe_integration)
  if(TARGET ${AHE_TARGET})
   add_custom_command(TARGET ${AHE_TARGET} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${AHE_ASAN_RUNTIME}" "$<TARGET_FILE_DIR:${AHE_TARGET}>" VERBATIM)
  endif()
 endforeach()
endif()
install(TARGETS application-hang-explainer RUNTIME DESTINATION bin)
install(FILES README.md LICENSE VALIDATION.md DESTINATION share/application-hang-explainer)
